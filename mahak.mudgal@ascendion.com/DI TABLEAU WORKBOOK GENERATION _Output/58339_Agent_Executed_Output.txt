          <relation type='table' name='[Custom SQL Query - data_state_fact]'>
            <text>select
      *,
      date_trunc(
        hour,
        coalesce(INSERT_TIMESTAMP_JMS_EVENT, INSERT_TIMESTAMP_PARKING_EVENT,
                INSERT_TIMESTAMP_PARKING_STOPPED_EVENT, INSERT_TIMESTAMP_TRANSACTION_FACT)
      ) as CREATE_HOUR,
      case
        when INSERT_TIMESTAMP_TRANSACTION_FACT is not null then &apos;PRESENT_IN_TRANSACTION_FACT&apos;
        -- JMS and parking event are present, but reporting data has not present in Transaction Fact for 3 or more hours
        when INSERT_TIMESTAMP_JMS_EVENT is not null
            and coalesce(INSERT_TIMESTAMP_PARKING_EVENT, INSERT_TIMESTAMP_PARKING_STOPPED_EVENT) is not null
            and INSERT_TIMESTAMP_TRANSACTION_FACT is null
            and timediff(hour, CREATE_HOUR, current_timestamp()) &gt;= 3 then &apos;MISSING_IN_TRANSACTION_FACT_3_OR_MORE_HOURS&apos;
        -- JMS and parking event are present, but reporting data has not present in Transaction Fact for under 3 hours
        when INSERT_TIMESTAMP_JMS_EVENT is not null
            and coalesce(INSERT_TIMESTAMP_PARKING_EVENT, INSERT_TIMESTAMP_PARKING_STOPPED_EVENT) is not null
            and INSERT_TIMESTAMP_TRANSACTION_FACT is null
            and timediff(hour, CREATE_HOUR, current_timestamp()) &lt; 3 then &apos;MISSING_IN_TRANSACTION_FACT_LESS_THAN_3_HOURS&apos;
        -- JMS event is present, but associated parking event has not present for 3 or more hours
        when timediff(hour, INSERT_TIMESTAMP_JMS_EVENT, current_timestamp()) &gt;= 3
            and coalesce(INSERT_TIMESTAMP_PARKING_EVENT, INSERT_TIMESTAMP_PARKING_STOPPED_EVENT) is null
            and INSERT_TIMESTAMP_TRANSACTION_FACT is null then &apos;MISSING_IN_TRANSACTION_FACT_PENDING_PARKING_EVENT_3_OR_MORE_HOURS&apos;
        -- JMS event is present, but associated parking event has not present for under 3 hours
        when timediff(hour, INSERT_TIMESTAMP_JMS_EVENT, current_timestamp()) &lt; 3
            and coalesce(INSERT_TIMESTAMP_PARKING_EVENT, INSERT_TIMESTAMP_PARKING_STOPPED_EVENT) is null
            and INSERT_TIMESTAMP_TRANSACTION_FACT is null then &apos;MISSING_IN_TRANSACTION_FACT_PENDING_PARKING_EVENT_LESS_THAN_3_HOURS&apos;
        -- Parking event is present, but associated JMS event has not present for 3 or more hours
        when timediff(hour, coalesce(INSERT_TIMESTAMP_PARKING_EVENT, INSERT_TIMESTAMP_PARKING_STOPPED_EVENT), current_timestamp()) &gt;= 3
            and INSERT_TIMESTAMP_JMS_EVENT is null
            and INSERT_TIMESTAMP_TRANSACTION_FACT is null then &apos;MISSING_IN_TRANSACTION_FACT_PENDING_JMS_EVENT_3_OR_MORE_HOURS&apos;
        -- Parking event is present, but associated JMS event has not present for under 3 hours
        when timediff(hour, coalesce(INSERT_TIMESTAMP_PARKING_EVENT, INSERT_TIMESTAMP_PARKING_STOPPED_EVENT), current_timestamp()) &lt; 3
            and INSERT_TIMESTAMP_JMS_EVENT is null
            and INSERT_TIMESTAMP_TRANSACTION_FACT is null then &apos;MISSING_IN_TRANSACTION_FACT_PENDING_JMS_EVENT_LESS_THAN_3_HOURS&apos;
      end as DATA_STATE
    from REPORTING_DBT.REPORTING_DATA_STATE_FACT</text>
